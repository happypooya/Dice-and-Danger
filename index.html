function consumeDiceForSum(sumVal, isDoubleReq, isDoubleDoor) { const valid = getValidPairs(poolDice); let match = null; if (selectedDiceIds.length === 2) { match = valid.find(p => { if (isDoubleDoor || isDoubleReq) return p.isDouble && (!sumVal || p.sum === sumVal); if (sumVal && p.sum === sumVal) return true; return false; }); } else { if (isDoubleDoor || isDoubleReq) { match = valid.find(p => p.isDouble && (!sumVal || p.sum === sumVal) && !p.usesBlack) || valid.find(p => p.isDouble && (!sumVal || p.sum === sumVal)); } else { match = valid.find(p => p.sum === sumVal && !p.usesBlack) || valid.find(p => p.sum === sumVal); } } if (match) { const removedIds = match.diceObjects.map(d => d.id); poolDice = poolDice.filter(d => !removedIds.includes(d.id)); selectedDiceIds = []; return match.diceObjects; } return null; }
